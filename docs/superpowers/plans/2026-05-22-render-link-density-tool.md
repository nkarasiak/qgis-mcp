# `qgis_render_link_density` Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `qgis_render_link_density` — the v2 tool flagged in `docs/DESIGN.md` §8 open question #8 — and its one-time prep script `scripts/build_drm_network.py`. The tool aggregates PFLOW trajectory CSVs by `link_id`, joins to the pre-built DRM road network, and renders a graduated line-density choropleth.

**Architecture:** MCP-side aggregation, plugin-side rendering — matches the v0.3 choropleth pattern. The prep script `scripts/build_drm_network.py` consumes the 47 prefecture-sharded DRM TSVs (~14 GB total) and writes a single `assets/drm_network.gpkg` indexed by `link_id`. The MCP tool streams trajectory CSVs with stdlib `csv`, builds `{link_id → density}` MCP-side, and dispatches `render_link_density` to the plugin with the small aggregated dict + a path to the GeoPackage.

**Tech Stack:** Python 3.12, `uv`, FastMCP, PyQGIS (plugin-side). New optional dependency: `pyogrio>=0.7` as the `[drm]` extra, used only by the prep script — the runtime tool has no new deps.

**Prereq:** Plan 1 (rename to `qgis-mcp-workflows`) must be merged before this plan starts. Every file path, import, and env var below assumes the post-rename names.

**Scope decisions (locked in before this plan was written):**
- Aggregation is **MCP-side, streaming** (stdlib `csv`, dict accumulator). Trajectory CSVs are never loaded fully into memory.
- DRM prep is **one-time, pre-built** via `scripts/build_drm_network.py` (analogous to `scripts/build_scaled_zones.py`). Tool runtime never parses DRM TSVs.
- The prep script uses **pyogrio** (new `[drm]` extra). Adds ~50 MB to the optional install footprint.
- Output asset: `assets/drm_network.gpkg`. Single GeoPackage, one feature per link, EPSG:4326, indexed on `link_id`. Built once per DRM-vintage update.
- Aggregations supported: `"count"` (default — points per link) and `"sum"` (sum a numeric column per link).
- New error: `DRMNetworkNotFoundError`. Reuses `FieldNotFoundError`, `JoinError`, `EmptyAfterFilterError`.

```python
"""Build assets/drm_network.gpkg from per-prefecture DRM TSVs.

DRM (Digital Road Map) source: H:\\Dropbox\\PFLOW\\data\\network\\drm_NN.tsv
(NN = 01..47 by prefecture, ~14 GB total, EPSG:4326, tab-separated, no header).

Schema per DESIGN.md §10:
    1=link_id, 2=from_node, 3=to_node, 4=road_class_code, 5-8=c5..c8 (opaque),
    9=from_lon, 10=from_lat, 11=to_lon, 12=to_lat, 13/14=wkt_linestring

Output: assets/drm_network.gpkg — single GeoPackage, all prefectures combined,
indexed by link_id. Used as the static layer for qgis_render_link_density.

Run once (or whenever DRM is updated):
    uv run --no-sync --extra drm scripts/build_drm_network.py \\
        --drm-dir 'H:/Dropbox/PFLOW/data/network' \\
        --output assets/drm_network.gpkg

Idempotent: regenerates in place. Skip-existing via --skip-existing.
"""

from __future__ import annotations

import argparse
import re
import sys
import time
from collections.abc import Iterator
from pathlib import Path

# WKT LINESTRING parser — matches DRM's column 14 format. Captures the
# space-separated lon/lat pairs inside the parens.
_LINESTRING_RE = re.compile(r"^\s*LINESTRING\s*\(\s*(.+?)\s*\)\s*$", re.IGNORECASE)


def parse_drm_tsv(path: Path) -> Iterator[dict]:
    """Stream-parse a DRM TSV. Yields one dict per non-blank line.

    Each dict has keys: link_id (str), road_class (str), from_lon (float),
    from_lat (float), to_lon (float), to_lat (float), coords (list[tuple[float, float]]).

    Raises ValueError on a non-LINESTRING WKT in column 14 — silent skip would
    mask DRM-vintage drift.
    """
    with path.open(encoding="utf-8", newline="") as f:
        for line_no, raw in enumerate(f, start=1):
            line = raw.rstrip("\n").rstrip("\r")
            if not line.strip():
                continue
            cols = line.split("\t")
            if len(cols) < 13:
                raise ValueError(
                    f"{path}:{line_no} expected >=13 tab-separated columns, got {len(cols)}"
                )
            wkt_col = cols[13] if len(cols) > 13 else cols[12]
            match = _LINESTRING_RE.match(wkt_col)
            if not match:
                raise ValueError(
                    f"{path}:{line_no} column 14 not a LINESTRING WKT: {wkt_col[:80]!r}"
                )
            coords: list[tuple[float, float]] = []
            for pair in match.group(1).split(","):
                parts = pair.strip().split()
                if len(parts) != 2:
                    raise ValueError(
                        f"{path}:{line_no} malformed WKT coord pair: {pair!r}"
                    )
                lon, lat = float(parts[0]), float(parts[1])
                coords.append((lon, lat))
            yield {
                "link_id": cols[0],
                "road_class": cols[3],
                "from_lon": float(cols[8]),
                "from_lat": float(cols[9]),
                "to_lon": float(cols[10]),
                "to_lat": float(cols[11]),
                "coords": coords,
            }


def main() -> None:
    """CLI entry — placeholder for Task 4 (writes the GeoPackage)."""
    raise NotImplementedError("GeoPackage writing lands in Task 4 — parsing only for now.")


if __name__ == "__main__":
    main()
```

```python
def build_geopackage(
    tsv_paths: list[Path],
    output_path: Path,
    skip_existing: bool = False,
    progress_every: int = 100_000,
) -> dict:
    """Combine multiple DRM TSVs into a single GeoPackage.

    Returns a summary dict: {"n_links": int, "n_files": int, "elapsed_s": float}.
    """
    import pyogrio
    from shapely.geometry import LineString

    if skip_existing and output_path.exists():
        return {"n_links": 0, "n_files": 0, "elapsed_s": 0.0, "skipped": True}

    output_path.parent.mkdir(parents=True, exist_ok=True)
    if output_path.exists():
        output_path.unlink()

    t0 = time.monotonic()
    n_links = 0

    # pyogrio writes in batches via a dataframe; build per-file then append.
    # For ~14 GB DRM input, we accumulate ~500k features per prefecture and
    # write once per file rather than streaming row-by-row (avoids per-row
    # overhead). Memory: ~500k LineString features ≈ 500 MB peak — OK on a
    # dev machine, would need chunking for 16 GB-RAM-constrained CI.

    import geopandas as gpd  # bundled with pyogrio's deps

    write_kwargs = dict(driver="GPKG", layer="links")
    first_file = True

    for tsv_path in tsv_paths:
        print(f"  reading {tsv_path.name} ...", flush=True)
        link_ids: list[str] = []
        road_classes: list[str] = []
        geoms: list[LineString] = []
        for row in parse_drm_tsv(tsv_path):
            link_ids.append(row["link_id"])
            road_classes.append(row["road_class"])
            geoms.append(LineString(row["coords"]))
            n_links += 1
            if n_links % progress_every == 0:
                print(f"    {n_links:,} links parsed", flush=True)

        gdf = gpd.GeoDataFrame(
            {"link_id": link_ids, "road_class": road_classes},
            geometry=geoms,
            crs="EPSG:4326",
        )
        if first_file:
            pyogrio.write_dataframe(gdf, output_path, append=False, **write_kwargs)
            first_file = False
        else:
            pyogrio.write_dataframe(gdf, output_path, append=True, **write_kwargs)

    elapsed = time.monotonic() - t0
    print(f"  wrote {n_links:,} links to {output_path} in {elapsed:.1f}s", flush=True)
    return {"n_links": n_links, "n_files": len(tsv_paths), "elapsed_s": elapsed}


def main() -> None:
    parser = argparse.ArgumentParser(prog="build_drm_network")
    parser.add_argument(
        "--drm-dir",
        default="H:/Dropbox/PFLOW/data/network",
        help="Directory containing drm_NN.tsv files (default: H:/Dropbox/PFLOW/data/network).",
    )
    parser.add_argument(
        "--output",
        default="assets/drm_network.gpkg",
        help="Output GeoPackage path (default: assets/drm_network.gpkg).",
    )
    parser.add_argument(
        "--pattern",
        default="drm_*.tsv",
        help="Glob pattern for input files (default: drm_*.tsv).",
    )
    parser.add_argument(
        "--skip-existing",
        action="store_true",
        help="Exit successfully if output already exists.",
    )
    args = parser.parse_args()

    drm_dir = Path(args.drm_dir)
    output = Path(args.output).resolve()

    tsv_paths = sorted(drm_dir.glob(args.pattern))
    if not tsv_paths:
        sys.exit(f"No files matching {args.pattern!r} in {drm_dir}")

    print(f"Found {len(tsv_paths)} TSV files in {drm_dir}", flush=True)
    summary = build_geopackage(
        tsv_paths=tsv_paths,
        output_path=output,
        skip_existing=args.skip_existing,
    )
    if summary.get("skipped"):
        print(f"Skipped — {output} already exists.")
        return
    print(
        f"Done: {summary['n_links']:,} links from {summary['n_files']} prefectures "
        f"→ {output} ({summary['elapsed_s']:.1f}s)."
    )
```

```
    if aggregation == "sum" and value_col is None:

    if aggregation not in ("count", "sum"):

    density: dict[str, float] = {}
    n_rows = 0

    for path in csv_paths:
        with open(path, encoding="utf-8", newline="") as f:
            reader = _csv.DictReader(f)
            columns = reader.fieldnames or []

                if not link_id:

                    except (TypeError, ValueError):

                    if val != val:  # NaN check
                        continue
                    density[link_id] = density.get(link_id, 0.0) + val
```

```python
def render_link_density(
    self,
    density: dict,
    drm_network_path: str,
    output_png: str,
    link_id_col: str,
    aggregation: str,
    value_col: str | None,
    n_classes: int,
    mode: str,
    palette: str,
    extent: list | None,
    basemap_paths: list,
    width: int,
    height: int,
    dpi: int,
) -> dict:
    """Render a graduated link-density choropleth from a {link_id → density} dict.

    Steps:
        1. Load the DRM line layer from drm_network_path.
        2. Add a numeric attribute `density_field`, populate from the dict.
        3. Mark unmatched links with density=NULL (or filter them out).
        4. Apply graduated renderer with the chosen breaks.
        5. Render with basemaps below, links above.
        6. Save PNG, clean up.
    """
    from qgis.core import (
        QgsClassificationEqualInterval,
        QgsClassificationJenks,
        QgsClassificationPrettyBreaks,
        QgsClassificationQuantile,
        QgsCoordinateReferenceSystem,
        QgsField,
        QgsGraduatedSymbolRenderer,
        QgsLineSymbol,
        QgsMapRendererParallelJob,
        QgsMapSettings,
        QgsProject,
        QgsRectangle,
        QgsStyle,
        QgsVectorLayer,
    )
    from qgis.PyQt.QtCore import QSize, QVariant
    from qgis.PyQt.QtGui import QColor, QImage, QPainter

    density_field = "n_points" if aggregation == "count" else f"sum_{value_col}"
    crs_4326 = QgsCoordinateReferenceSystem("EPSG:4326")

    # Load DRM line layer
    drm_layer = QgsVectorLayer(drm_network_path, "drm_network", "ogr")
    if not drm_layer.isValid():
        raise RuntimeError(f"DRM_LOAD_FAILED: {drm_network_path}")

    # Add density attribute and populate via link_id join
    drm_layer.startEditing()
    drm_layer.dataProvider().addAttributes([QgsField(density_field, QVariant.Double)])
    drm_layer.updateFields()
    fid_to_density: list[tuple[int, float]] = []
    n_matched = 0
    field_idx = drm_layer.fields().indexFromName(density_field)
    for feat in drm_layer.getFeatures():
        link_id = str(feat["link_id"])
        if link_id in density:
            fid_to_density.append((feat.id(), density[link_id]))
            n_matched += 1
    attr_map = {fid: {field_idx: val} for fid, val in fid_to_density}
    drm_layer.dataProvider().changeAttributeValues(attr_map)
    drm_layer.commitChanges()

    n_unmatched_link_ids = len(density) - n_matched

    # Filter to rendered subset: only features with density set
    drm_layer.setSubsetString(f'"{density_field}" IS NOT NULL')
    n_links_rendered = drm_layer.featureCount()

    # Graduated renderer
    mode_map = {
        "quantile": QgsClassificationQuantile(),
        "equal_interval": QgsClassificationEqualInterval(),
        "natural_breaks": QgsClassificationJenks(),
        "pretty": QgsClassificationPrettyBreaks(),
    }
    method = mode_map.get(mode, QgsClassificationQuantile())
    style = QgsStyle.defaultStyle()
    color_ramp = style.colorRamp(palette) if palette in style.colorRampNames() else style.colorRamp("YlOrRd")

    renderer = QgsGraduatedSymbolRenderer.createRenderer(
        vlayer=drm_layer,
        attrName=density_field,
        classes=n_classes,
        mode=QgsGraduatedSymbolRenderer.Custom,  # we supply breaks via method
        symbol=QgsLineSymbol.createSimple({"line_width": "1.0", "color": "black"}),
        ramp=color_ramp,
        legendFormat=QgsGraduatedSymbolRenderer.RangeLowerLabel("%1 - %2"),
    )
    # Apply the chosen classification method
    renderer.setClassificationMethod(method)
    renderer.updateClasses(drm_layer, n_classes)

    # Data-defined line width: scale 0.5 → 3.0 mm with density
    breaks = [r.upperValue() for r in renderer.ranges()]
    min_v = min(density.values())
    max_v = max(density.values())
    drm_layer.setRenderer(renderer)

    # Build basemap layers
    extra_layers = []
    for bp in basemap_paths:
        bl = QgsVectorLayer(bp, "basemap", "ogr")
        if bl.isValid():
            extra_layers.append(bl)

    # Render
    settings = QgsMapSettings()
    settings.setBackgroundColor(QColor("white"))
    settings.setOutputSize(QSize(width, height))
    settings.setOutputDpi(dpi)
    settings.setDestinationCrs(crs_4326)
    layer_stack = list(extra_layers) + [drm_layer]
    settings.setLayers(layer_stack)
    if extent is not None:
        rect = QgsRectangle(extent[0], extent[1], extent[2], extent[3])
    else:
        rect = drm_layer.extent()
        # 5% pad
        dx, dy = rect.width() * 0.05, rect.height() * 0.05
        rect = QgsRectangle(rect.xMinimum() - dx, rect.yMinimum() - dy,
                            rect.xMaximum() + dx, rect.yMaximum() + dy)
    settings.setExtent(rect)

    img = QImage(width, height, QImage.Format_ARGB32_Premultiplied)
    img.fill(QColor("white"))
    painter = QPainter(img)
    job = QgsMapRendererParallelJob(settings)
    job.start()
    job.waitForFinished()
    rendered = job.renderedImage()
    painter.drawImage(0, 0, rendered)
    painter.end()
    img.save(output_png, "PNG")

    # Cleanup: revert subset, drop the temporary attribute, unload
    drm_layer.setSubsetString("")
    drm_layer.startEditing()
    drm_layer.dataProvider().deleteAttributes([field_idx])
    drm_layer.commitChanges()
    QgsProject.instance().removeMapLayer(drm_layer.id())

    return {
        "output_path": output_png,
        "width": width, "height": height, "dpi": dpi,
        "extent": [rect.xMinimum(), rect.yMinimum(), rect.xMaximum(), rect.yMaximum()],
        "crs": "EPSG:4326",
        "n_layers": len(layer_stack),
        "n_links_with_traffic": len(density),
        "n_links_rendered": n_links_rendered,
        "n_unmatched_link_ids": n_unmatched_link_ids,
        "density_field": density_field,
        "breaks": breaks,
        "mode": mode,
        "min_density": min_v,
        "max_density": max_v,
    }
```
